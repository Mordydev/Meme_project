## YOLO Mode Decision Tree

```python
# Pseudo-code for decision logic

def yolo_production(user_input):
    # Phase 1: Concept
    if has_product:
        content_type = "transformation"
        keywords = ["premium", "elegant"]
    elif has_story:
        content_type = "character"
        keywords = ["authentic", "emotional"]
    else:
        content_type = "compilation"
        keywords = ["dynamic", "variety"]
    
    # Phase 2: Visual
    if content_type == "transformation":
        camera = "macro_to_wide"
        lighting = "evolving"
    elif content_type == "character":
        camera = "intimate_50mm"
        lighting = "natural"
    else:
        camera = "varied"
        lighting = "mixed"
    
    # Phase 3: Script
    word_limit = 15 if single_speaker else 10
    dialogue = generate_natural(word_limit)
    
    # Phase 4: Platform
    if platform == "tiktok":
        hook_time = 1.5
        variation = "high_energy"
    elif platform == "youtube":
        hook_time = 2.0
        variation = "balanced"
    else:
        hook_time = 1.0
        variation = "aesthetic"
    
    # Phase 5: Enhancement
    apply_polish(preserve_working=True)
    
    # Phase 6: Validation
    score = run_checklist()
    if score < 95:
        auto_fix()
        revalidate()
    
    return final_output
```
