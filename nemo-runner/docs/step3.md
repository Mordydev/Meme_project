        // src/lib/game/shaders/common/caustics.common.glsl.ts
        // Reuses noise functions, assuming they are available via #include or directly in the final shader
        // #include <noise2D> // or similar if noise.glsl.ts is structured for it

        export const causticPattern = `
        // Ensure random2D and valueNoise2D are included or defined before this chunk

        float FBM(vec2 p, int octaves, float persistence, float lacunarity) {
            float total = 0.0;
            float frequency = 1.0;
            float amplitude = 1.0;
            float maxValue = 0.0; // Used for normalizing result to 0.0 - 1.0
            for(int i = 0; i < octaves; i++) {
                total += valueNoise2D(p * frequency) * amplitude;
                maxValue += amplitude;
                amplitude *= persistence;
                frequency *= lacunarity;
            }
            return total / maxValue;
        }

        vec3 getCausticColor(vec3 worldPosition, float time, float scale, float intensity, vec3 causticBaseColor) {
            // Multiple layers of scrolling noise for a watery caustic effect
            vec2 uv1 = worldPosition.xz / scale + time * 0.05; // Slower scroll for base layer
            vec2 uv2 = worldPosition.xz / (scale * 0.6) + time * 0.08; // Faster, smaller details
            vec2 uv3 = worldPosition.xz / (scale * 0.3) + time * 0.12;

            float pattern1 = FBM(uv1, 3, 0.5, 2.0); // 3 octaves FBM
            float pattern2 = FBM(uv2, 2, 0.5, 2.0); // 2 octaves FBM for sharper details
            float pattern3 = FBM(uv3, 2, 0.4, 2.5); // Even smaller details

            // Combine patterns - multiply to get bright intersections, then sharpen
            float combinedPattern = pattern1 * pattern2 * pattern3;
            combinedPattern = pow(combinedPattern, 3.0); // Sharpen the bright spots
            combinedPattern = smoothstep(0.1, 0.6, combinedPattern); // Adjust contrast

            // Add some subtle distortion to the main pattern using another noise layer
            vec2 distortUv = worldPosition.xz / (scale * 2.0) + time * 0.03;
            float distortion = valueNoise2D(distortUv) * 0.3 - 0.15; // Small distortion amount
            combinedPattern *= (1.0 + distortion);
            
            combinedPattern = clamp(combinedPattern, 0.0, 1.0);

            return causticBaseColor * combinedPattern * intensity;
        }
        `;
        const CausticsGLSL = { causticPattern };
        export default CausticsGLSL;

            *   Includes `noise2D` and `causticPattern`.
            *   Uses `uBaseColor` for sand, adds a subtle noise pattern to it.
            *   Performs basic Lambertian lighting (can be expanded to full PBR later if needed).
            *   Calls `getCausticColor` and adds it to the lit base color.
            *   Uniforms for caustic parameters are defined, to be set by `ShaderManager` using defaults from `configSystem`.
        *   `ShaderManager` registers the `causticPattern` chunk and the `seafloorShader`.
        *   `SeafloorAsset` now requests `seafloorShader` from `ShaderManager`.
*   **I. Proactive Considerations & NFR Alignment for this Sub-Task:**
    *   **Caustic Performance:** The FBM in GLSL can be expensive. The number of octaves (currently 3 and 2) and other parameters will need tuning for performance vs. quality.
    *   **Shader Complexity:** The seafloor shader is kept relatively simple for now, focusing on diffuse lighting + caustics. More advanced PBR shading would add complexity.
    *   **Uniforms:** Ensure all uniforms used in GLSL are correctly provided by `ShaderManager` (globals like `uTime`) or through the `defaultUniforms` of the registered shader (like `uCausticIntensity`, `uCausticScale` etc.).

### 5. Troubleshooting Common Issues & Proactive Error Prevention for THIS STEP

*   **Symptom:** Caustics are not visible or not animating.
    *   **Likely Cause(s):** Seafloor material not using the correct custom shader; `uTime` uniform not reaching/being used in the caustic shader; caustic intensity too low or color blending poorly; incorrect world position calculation in shader; `LightingManager.update()` not called or not updating relevant uniforms.
    *   **Solution/Debugging Tip:** Use Spector.js to inspect the seafloor material and its shader code/uniforms. Output intermediate values from the caustic shader (e.g., `gl_FragColor = vec4(vWorldPosition.xxx, 0.0, 1.0);` or `gl_FragColor = vec4(noiseValue, noiseValue, noiseValue, 1.0);`). Verify `uTime` is updating in `ShaderManager`.
    *   **Prevention Note:** Start with very simple caustic logic (e.g., just scrolling noise) and build complexity. Ensure uniforms are correctly named and typed.
*   **Symptom:** Fog is too dense, not visible, or starts/ends at weird distances.
    *   **Likely Cause(s):** Incorrect `fogNearFactor`/`fogFarFactor` in `gameConfig.ts`; `viewDepth` assumption in `LightingManager.initializeFog()` doesn't match actual camera perspective; fog color identical to sky making it hard to perceive depth difference.
    *   **Solution/Debugging Tip:** Temporarily use very distinct fog color. Adjust `fogNearFactor` and `fogFarFactor` in `gameConfig.ts` with large steps to see their effect. Log the calculated `near` and `far` values in `LightingManager`.
    *   **Prevention Note:** Make fog parameters highly configurable. Test with a camera that moves through the scene to observe fog transitions.
*   **Symptom:** Performance drop after enabling caustics.
    *   **Likely Cause(s):** Caustic shader (FBM calculation) is too complex (too many octaves, expensive noise functions).
    *   **Solution/Debugging Tip:** Reduce FBM octaves in `caustics.common.glsl.ts`. Simplify noise functions. Profile GPU usage in browser dev tools.
    *   **Prevention Note:** Implement caustics iteratively, checking performance at each step. Provide quality settings in `gameConfig.ts` to disable or simplify caustics on lower-end targets if necessary later.
